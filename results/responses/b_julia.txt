"""
    newton_raphson(f, df, x0, tol, max_iter)

Newton-Raphson method for finding roots of a function.

# Arguments
- `f`: The function whose root is sought.
- `df`: The derivative of `f`.
- `x0`: Initial guess for the root.
- `tol`: Tolerance for convergence (absolute error).
- `max_iter`: Maximum number of iterations allowed.

# Returns
- The approximate root found, or `nothing` if the method fails.
"""
function newton_raphson(f, df, x0::Number, tol::Number=1e-8, max_iter::Int=100)::Number
    x = x0
    for i in 1:max_iter
        fx = f(x)
        dfx = df(x)
        # Avoid division by zero
        if abs(dfx) < tol
            error("Derivative too close to zero at iteration $i; method fails.")
        end
        x_new = x - fx / dfx
        error_abs = abs(x_new - x)
        println("Iteration $i: x = $(x), f(x) = $(fx), error = $(error_abs)")
        if error_abs < tol
            return x_new
        end
        x = x_new
    end
    error("Failed to converge within $max_iter iterations.")
end

# Define the function and its derivative
f(x) = x^3 - 2x - 5
df(x) = 3x^2 - 2

# Run the method
root = newton_raphson(f, df, 2.0)
println("Root: ", root)

**Explanation:**
- The Newton-Raphson method iteratively improves the guess using the formula:  
  `x_{n+1} = x_n - f(x_n) / f'(x_n)`
- The loop prints the iteration number, current `x`, `f(x)`, and the absolute error between successive approximations.
- Convergence is checked against the tolerance `tol`. If the error falls below `tol`, the method stops early.
- The derivative must not be near zero to avoid division by zero.
