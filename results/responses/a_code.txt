from typing import List

def merge_sort(nums: List[int]) -> List[int]:
    """
    Sort a list of integers in ascending order using the merge sort algorithm.

    Args:
        nums (List[int]): The list of integers to sort.

    Returns:
        List[int]: A new list containing the sorted integers.

    Raises:
        TypeError: If the input is not a list.

    Edge cases handled:
        - Empty list returns an empty list.
        - Single-element list returns a new list with the same element.
    """
    if not isinstance(nums, list):
        raise TypeError("Input must be a list.")

    # Base case: a list with 0 or 1 element is already sorted.
    if len(nums) <= 1:
        return nums.copy()

    # Split the list into two halves.
    mid = len(nums) // 2
    left_half = merge_sort(nums[:mid])
    right_half = merge_sort(nums[mid:])

    # Merge the sorted halves.
    sorted_list = []
    i = j = 0

    while i < len(left_half) and j < len(right_half):
        if left_half[i] <= right_half[j]:
            sorted_list.append(left_half[i])
            i += 1
        else:
            sorted_list.append(right_half[j])
            j += 1

    # Append any remaining elements from either half.
    sorted_list.extend(left_half[i:])
    sorted_list.extend(right_half[j:])

    return sorted_list
