Here is the C program that implements a simple thread-safe queue using a linked list with pthreads.

```c
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>

// Define the node structure for the linked list
typedef struct Node {
    int data;
    struct Node* next;
} Node;

// Define the queue structure
typedef struct {
    Node* front;
    Node* rear;
    size_t size;
} Queue;

// Function to initialize the queue
void queue_init(Queue* q) {
    q->front = NULL;
    q->rear = NULL;
    q->size = 0;
}

// Function to push an element onto the queue
void queue_push(Queue* q, int value) {
    pthread_mutex_lock(&q_mutex);
    Node* new_node = (Node*)malloc(sizeof(Node));
    if (!new_node) {
        fprintf(stderr, "Memory allocation failed\n");
        return;
    }
    new_node->data = value;
    new_node->next = NULL;

    if (q->rear == NULL) {
        q->front = new_node;
    } else {
        q->rear->next = new_node;
    }
    q->rear = new_node;
    q->size++;
    pthread_mutex_unlock(&q_mutex);
}

// Function to pop an element from the queue
int queue_pop(Queue* q) {
    int value;
    pthread_mutex_lock(&q_mutex);
    if (q->front == NULL) {
        pthread_mutex_unlock(&q_mutex);
        fprintf(stderr, "Queue is empty\n");
        return -1;
    }
    value = q->front->data;
    Node* temp = q->front;
    q->front = q->front->next;
    if (q->front == NULL) {
        q->rear = NULL;
    }
    pthread_mutex_unlock(&q_mutex);
    free(temp);
    q->size--;
    return value;
}

// Function to get the size of the queue
size_t queue_size(Queue* q) {
    pthread_mutex_lock(&q_mutex);
    size_t size = q->size;
    pthread_mutex_unlock(&q_mutex);
    return size;
}

// Function to destroy the queue
void queue_destroy(Queue* q) {
    pthread_mutex_lock(&q_mutex);
    while (q->front != NULL) {
        Node* temp = q->front;
        q->front = q->front->next;
        free(temp);
    }
    q->front = NULL;
    q->rear = NULL;
    q->size = 0;
    pthread_mutex_unlock(&q_mutex);
}

// Global mutex for thread safety
pthread_mutex_t q_mutex = PTHREAD_MUTEX_INITIALIZER();

int main() {
    Queue q;
    queue_init(&q);

    // Push 5 integers
    for (int i = 0; i < 5; i++) {
        queue_push(&q, i);
    }

    // Pop all elements
    while (queue_size(&q) > 0) {
        int val = queue_pop(&q);
        if (val == -1) {
            fprintf(stderr, "Queue is empty\n");
            break;
        }
        printf("%d\n", val);
    }

    queue_destroy(&q);
    return 0;
}
```

### Key Features:
1. **Thread Safety**: Uses `pthread_mutex` to ensure safe access to the queue from multiple threads.
2. **Memory Management**: Properly allocates and frees memory for nodes.
3. **Error Handling**: Checks for memory allocation failures and queue emptiness.
4. **Demonstration**: The `main` function shows initializing the queue, pushing 5 integers, and popping them all.
