<canvas id="game-canvas" width="500" height="500"></canvas>

<script>
    const canvas = document.getElementById('game-canvas');
    const ctx = canvas.getContext('2d');
    const scoreElement = document.getElementById('score');
    const timerElement = document.getElementById('timer');
    const gameOverElement = document.getElementById('game-over');
    const startBtn = document.getElementById('start-btn');

    let score = 0;
    let timer = 30000;
    let gameInterval;
    let gameActive = false;

    function drawTarget() {
        ctx.clearRect(0, 0, canvas.width, canvas.height);
        const x = Math.random() * canvas.width;
        const y = Math.random() * canvas.height;
        const radius = Math.min(canvas.width, canvas.height) / 2;
        ctx.beginPath();
        ctx.arc(x, y, radius, 0, Math.PI * 2);
        ctx.fillStyle = '#3498db';
        ctx.fill();
        ctx.strokeStyle = '#3498db';
        ctx.stroke();
    }

    function drawScore() {
        scoreElement.textContent = `Score: ${score}`;
    }

    function drawTimer() {
        timerElement.textContent = `Timer: ${Math.max(0, timer)}ms`;
    }

    function drawGameOver() {
        gameOverElement.textContent = `Game Over! Your score was: ${score}`;
        gameOverElement.style.color = '#e74c3c';
        startBtn.disabled = true;
        startBtn.style.backgroundColor = '#e74c3c';
    }

    function startGame() {
        if (!gameActive) {
            score = 0;
            timer = 30000;
            gameActive = true;
            gameInterval = setInterval(() => {
                timer -= 1000;
                drawTimer();
                if (timer <= 0) {
                    clearInterval(gameInterval);
                    gameActive = false;
                    drawGameOver();
                }
            }, 1000);
            drawTarget();
            drawScore();
            drawTimer();
            startBtn.disabled = true;
            startBtn.textContent = 'Game Over';
        }
    }

    function handleClick(e) {
        if (!gameActive) return;
        const rect = canvas.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        const radius = Math.min(canvas.width, canvas.height) / 2;
        if (x <= radius && x >= -radius && y <= radius && y >= -radius) {
            score += 10;
            drawTarget();
            drawScore();
        }
    }

    canvas.addEventListener('click', handleClick);
    startBtn.addEventListener('click', startGame);

    document.getElementById('start-btn').textContent = 'Start Game';
</script>

</body>
